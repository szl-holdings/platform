  id SERIAL PRIMARY KEY,
  name TEXT,
  type TEXT,
  source TEXT,
  owner TEXT,
  score INT,
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE deals (
  id SERIAL PRIMARY KEY,
  property TEXT,
  stage TEXT,
  owner TEXT,
  probability INT,
  risk INT,
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE distress_properties (
  id SERIAL PRIMARY KEY,
  address TEXT,
  borough TEXT,
  type TEXT,
  stage TEXT,
  score INT,
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE vulnerabilities (
  id SERIAL PRIMARY KEY,
  asset TEXT,
  severity TEXT,
  status TEXT,
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE incidents (
  id SERIAL PRIMARY KEY,
  type TEXT,
  severity TEXT,
  status TEXT,
  created_at TIMESTAMP DEFAULT NOW()

STEP 4 — SERVER (API)
==================================================

CREATE /server/src/index.ts:

import express from "express";
import cors from "cors";
import { Pool } from "pg";

const app = express();
app.use(cors());
app.use(express.json());

const pool = new Pool({
  connectionString: process.env.DATABASE_URL || "postgres://szl:szl@localhost:5432/szl"
});

app.get("/api/health", (_, res) => res.json({ status: "ok" }));

// TERRA ROUTES
app.get("/api/terra/leads", async (_, res) => {
  const { rows } = await pool.query("SELECT * FROM leads LIMIT 50");
  res.json(rows);
});

app.get("/api/terra/deals", async (_, res) => {
  const { rows } = await pool.query("SELECT * FROM deals LIMIT 50");
  res.json(rows);
});

app.get("/api/terra/distress", async (_, res) => {
  const { rows } = await pool.query("SELECT * FROM distress_properties LIMIT 50");
  res.json(rows);
});

// FIRESTORM ROUTES
app.get("/api/firestorm/vulnerabilities", async (_, res) => {
  const { rows } = await pool.query("SELECT * FROM vulnerabilities LIMIT 50");
  res.json(rows);
});

app.get("/api/firestorm/incidents", async (_, res) => {
  const { rows } = await pool.query("SELECT * FROM incidents LIMIT 50");
  res.json(rows);
});

app.listen(4000, () => console.log("Server running on 4000"));

STEP 5 — SEED DATA

CREATE /server/src/db/seed.ts:

import { Pool } from "pg";

const pool = new Pool({
  connectionString: "postgres://szl:szl@localhost:5432/szl"
});

async function seed() {
  await pool.query("INSERT INTO leads (name,type,source,owner,score) VALUES ('John Doe','buyer','web','agent1',80)");
  await pool.query("INSERT INTO deals (property,stage,owner,probability,risk) VALUES ('123 Main St','offer','agent1',70,30)");
  await pool.query("INSERT INTO distress_properties (address,borough,type,stage,score) VALUES ('456 Bronx Ave','Bronx','foreclosure','pre',85)");
  await pool.query("INSERT INTO vulnerabilities (asset,severity,status) VALUES ('server-1','high','open')");
  await pool.query("INSERT INTO incidents (type,severity,status) VALUES ('login anomaly','medium','open')");
  console.log("Seeded");
  process.exit();
}

seed();

CREATE Vite React app in /apps/terra

CREATE App.tsx:

import { useEffect, useState } from "react";

export default function App() {
  const [deals, setDeals] = useState([]);

  useEffect(() => {
    fetch("http://localhost:4000/api/terra/deals")
      .then(r => r.json())
      .then(setDeals);
  }, []);

  return (
    <div style={{ padding: 20 }}>
      <h1>Terra — Deal Pipeline</h1>
      {deals.map((d:any) => (
        <div key={d.id}>
          {d.property} — {d.stage} — Risk: {d.risk}
        </div>
      ))}
    </div>
  );
}
